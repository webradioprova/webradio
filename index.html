async function updateNowPlaying() { try { const response = await fetch('https://eu8.fastcast4u.com/rpc/webradioprova/streaminfo.get'); const data = await response.json(); if (data && data.type === 'result' && data.data && data.data[0]) { const info = data.data[0]; let artist = info.artist || ''; let title = info.title || ''; let album = info.album || ''; let rawSong = info.song || ''; // Se il server manda la scritta di default, mostriamo "Web Radio Prova in diretta" fissa if (rawSong.includes('FreeSHOUTCAST AutoDJ') || (artist.includes('FreeSHOUTCAST AutoDJ') && !title)) { let defaultDisplay = "Web Radio Prova in diretta"; if (defaultDisplay !== currentTrack) { currentTrack = defaultDisplay; trackTitleEl.textContent = defaultDisplay; document.title = `${defaultDisplay} | Web Radio Prova`; coverImg.src = DEFAULT_LOGO; } return; } if (!title && rawSong.includes(' - ')) { const parts = rawSong.split(' - '); if (!artist) artist = parts[0].trim(); title = parts.slice(1).join(' - ').trim(); } else if (!title) { title = rawSong; } if (artist && title.toLowerCase().startsWith(artist.toLowerCase())) { title = title.substring(artist.length).replace(/^[\s\-]+/, '').trim(); } let partsList = []; if (title) partsList.push(title); if (album) partsList.push(album); if (artist) partsList.push(artist); let fullDetails = partsList.join(' - '); if (!fullDetails) fullDetails = rawSong || 'Web Radio Prova'; if (fullDetails && fullDetails !== currentTrack) { currentTrack = fullDetails; trackTitleEl.textContent = fullDetails; document.title = `${fullDetails} | Web Radio Prova`; const lowerSong = fullDetails.toLowerCase(); if (lowerSong.includes('web radio') || lowerSong.includes('presentazione') || lowerSong.includes('station id') || lowerSong.includes('jingle')) { coverImg.src = DEFAULT_LOGO; } else { const coverUrl = await fetchCoverArt(title || fullDetails); coverImg.src = coverUrl; } } } } catch (error) { console.log('Errore info brano:', error); } }
