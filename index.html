async function updateNowPlaying() { try { const response = await fetch('https://eu8.fastcast4u.com/rpc/webradioprova/streaminfo.get'); const data = await response.json(); if (data && data.type === 'result' && data.data && data.data[0]) { const info = data.data[0]; let songString = info.song || ''; let album = info.album || ''; // Se la stringa contiene i dati, formattiamo con Titolo - Album - Artista if (songString.includes(' - ')) { const parts = songString.split(' - '); const artist = parts[0].trim(); const titleAndTags = parts.slice(1).join(' - ').trim(); // Se abbiamo l'album (o lo ricaviamo), lo inseriamo nel mezzo if (album && album !== '0' && album !== '') { songString = `${titleAndTags} - ${album} - ${artist}`; } else { songString = `${titleAndTags} - ${artist}`; } } if (songString && songString !== currentTrack) { currentTrack = songString; trackTitleEl.textContent = songString; document.title = `${songString} | Web Radio Prova`; const lowerSong = songString.toLowerCase(); if (lowerSong.includes('web radio') || lowerSong.includes('presentazione') || lowerSong.includes('station id') || lowerSong.includes('jingle')) { coverImg.src = DEFAULT_LOGO; } else { const coverUrl = await fetchCoverArt(songString); coverImg.src = coverUrl; } } } } catch (error) { console.log('Errore info brano:', error); } }
